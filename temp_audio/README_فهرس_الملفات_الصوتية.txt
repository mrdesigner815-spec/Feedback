  النص المنطوق بالسعودي: "ما شاء الله، تبارك الرحمن! شغل مميز وجهد مبارك يدل على تفوقك وإبداعك يا بطل."
  النص الأصلي من الدليل (صفحة 1): "مَا شَاءَ اللَّهُ، تَبَارَكَ الرَّحْمَنُ! عَمَلٌ مُتَمَيِّزٌ وَجُهْدٌ مُبَارَكٌ يَدُلُّ عَلَى تَفَوُّقِك."

القسم الأول: نماذج التغذية الراجعة للإجابة الصحيحة (التعزيز):
01. إِجَابَةٌ نَمُوذَجِيَّةٌ (صفحة 1)
   النص المنطوق بالسعودي: "ما شاء الله عليك! إجابة مضبوطة بالملي وصحيحة، كفو يا بطل وبيض الله وجهك، فخور جداً بهالمستوى العالي."
   النص الأصلي المشكول: "إِجَابَةٌ صَحِيحَةٌ وَدَقِيقَةٌ، مَا شَاءَ اللَّهُ! أَحْسَنْتَ التَّفْكِيرَ يَا بَطَلُ، وَأَنَا فَخُورٌ بِهَذَا الْمُسْتَوَى."

02. فَهْمٌ عَمِيقٌ (صفحة 1)
   النص المنطوق بالسعودي: "يا سلام عليك! إجابتك تبيّن قد إيش أنت فاهم الدرس ومستوعب كل التفاصيل بشكل ممتاز، أبدعت والله."
   النص الأصلي المشكول: "رَائِعٌ جِدًّا! إِجَابَتُكَ تَدُلُّ عَلَى فَهْمِكَ الْعَمِيقِ لِلدَّرْسِ وَاسْتِيعَابِكَ لِلْمَعْلُومَاتِ بِشَكْلٍ مُمْتَازٍ."

03. تَمَيُّزٌ مُسْتَمِرٌّ (صفحة 1)
   النص المنطوق بالسعودي: "ممتاز جداً! جبتها في الصميم وجبت الهدف بالتمام، واصل على هالمستوى والتفوق اللي يشرّف."
   النص الأصلي المشكول: "مُمْتَازٌ! لَقَدْ أَصَبْتَ الْهَدَفَ بِدِقَّةٍ. اسْتَمِرَّ عَلَى هَذَا التَّفَوُّقِ وَالنَّجَاحِ الْمُبْهِرِ."

04. تَشْجِيعٌ وَدَعْمٌ (صفحة 1)
   النص المنطوق بالسعودي: "تسلم يمينك وبارك الله بجهدك الطيب! ترتيبك للأفكار وصياغتك للجواب تفتح النفس ومثالية ما شاء الله."
   النص الأصلي المشكول: "بُورِكَتْ جُهُودُكَ الطَّيِّبَةُ. لَقَدْ أَبْدَعْتَ فِي تَرْتِيبِ الْأَفْكَارِ وَصِيَاغَةِ الْإِجَابَةِ بِشَكْلٍ مِثَالِيٍّ."

05. إِبْدَاعٌ لَامِعٌ (صفحة 1)
   النص المنطوق بالسعودي: "تفكير مبدع ورايق ما شاء الله! متميز بطرحك وفكرك الذكي، أسلوبك ينم عن ذكاء ولمعة موهبة حقيقية."
   النص الأصلي المشكول: "تَفْكِيرٌ مُبْدِعٌ وَخَلَّاقٌ! لَقَدْ تَمَيَّزْتَ بِهَذَا الطَّرْحِ الْجَمِيلِ الَّذِي يَدُلُّ عَلَى ذَكَائِكَ الْفَذِّ."

06. عَمَلٌ مُتْقَنٌ (صفحة 1)
   النص المنطوق بالسعودي: "ما شاء الله لا قوة إلا بالله! حلك للواجب يبيّن حرصك واهتمامك ودقتك العالية، الله يبارك فيك ويزيدك."
   النص الأصلي المشكول: "مَا شَاءَ اللَّهُ! إِجَابَتُكَ تَعْكِسُ مَدَى اهْتِمَامِكَ وَدِقَّتِكَ فِي حَلِّ الْوَاجِبَاتِ، بَارَكَ اللَّهُ فِيكَ."

القسم الثاني: نماذج التغذية الراجعة للإجابة الخاطئة (التوجيه والتصويب):
07. مُحَاوَلَةٌ جَيِّدَةٌ (صفحة 1)
   النص المنطوق بالسعودي: "محاولة شجاعة وحلوة منك يا بطل! بس ودّي نراجع الخطوة الثانية مرّة ثانية وتشوف كيف توصل للحل الصح بسهولة."
   النص الأصلي المشكول: "مُحَاوَلَةٌ شُجَاعَةٌ وَجَيِّدَةٌ! لَكِنْ أَرْجُو مِنْكَ مُرَاجَعَةَ الْخُطْوَةِ الثَّانِيَةِ مَرَّةً أُخْرَى لِتَصِلَ لِلصَّوَابِ."

08. تَوْجِيهٌ نَحْوَ الْحَلِّ (صفحة 1)
   النص المنطوق بالسعودي: "قريب جداً من الجواب الصح، باقيلك تكة صغيرة بس! وش رايك نفكر فيها من زاوية ثانية شوية؟"
   النص الأصلي المشكول: "لَقَدِ اقْتَرَبْتَ كَثِيرًا مِنَ الْإِجَابَةِ الصَّحِيحَةِ. هَلْ يُمْكِنُكَ التَّفْكِيرُ بِطَرِيقَةٍ مُخْتَلِفَةٍ قَلِيلًا؟"

09. دَعْمٌ عِنْدَ الْخَطَإِ (صفحة 2)
   النص المنطوق بالسعودي: "عادي جداً ولا يهمك، ما أحد يتعلم إلا بعد ما يجرّب ويخطي! ركّز حبتين على القاعدة الأساسية وأنا واثق إنك بتجيبها المرة الجاية."
   النص الأصلي المشكول: "لَا بَأْسَ بِالْخَطَإِ، فَهُوَ طَرِيقُنَا لِلتَّعَلُّمِ! رَكِّزْ أَكْثَرَ عَلَى الْقَاعِدَةِ الْأَسَاسِيَّةِ وَسَتَنْجَحُ حَتْمًا."

10. مُشَارَكَةٌ مُقَدَّرَةٌ (صفحة 2)
   النص المنطوق بالسعودي: "شاكر ومقدّر مشاركتك وتفاعلك الحلو بالصف. الجواب محتاج بس شوية تعديل بسيط، تعال نحاول فيها سوا من جديد."
   النص الأصلي المشكول: "أَشْكُرُ مُشَارَكَتَكَ وَتَفَاعُلَكَ الْمُمَيَّزَ. الْإِجَابَةُ تَحْتَاجُ إِلَى بَعْضِ التَّعْدِيلِ، دَعْنَا نُحَاوِلْ مَعًا مَرَّةً أُخْرَى."

11. خُطْوَةٌ نَحْوَ النَّجَاحِ (صفحة 2)
   النص المنطوق بالسعودي: "لا تشيل هم يا بطل ولا تتضايق، هذي الخطوة هي مفتاح فهمك للموضوع بشكل أعمق. يالله نعيد المحاولة سوا بروح عالية!"
   النص الأصلي المشكول: "لَا تُحْبَطْ يَا بَطَلُ، فَهَذِهِ الْعَثْرَةُ هِيَ مِفْتَاحُكَ لِفَهْمِ الْمَوْضُوعِ بِشَكْلٍ أَعْمَقَ. لِنُعِدِ الْمُحَاوَلَةَ مَعًا!"

12. بِحَاجَةٍ لِلْمُرَاجَعَةِ (صفحة 2)
   النص المنطوق بالسعودي: "في كلامك جزء صحيح وجميل، بس لو نرجع للدرس ونراجعه على السريع متأكد مية بالمية بنوصل للجواب الكامل والمثالي!"
   النص الأصلي المشكول: "إِجَابَتُكَ تَحْمِلُ جُزْءًا مِنَ الصِّحَّةِ، لَكِنْ لَوْ رَاجَعْنَا الدَّرْسَ مَرَّةً أُخْرَى سَنَصِلُ لِلْإِجَابَةِ الْمِثَالِيَّةِ بِالتَّأْكِيدِ."

القسم الثالث: نماذج عامة للبناء والتحفيز:
13. تَقَدُّمٌ مَلْحُوظٌ (صفحة 2)
   النص المنطوق بالسعودي: "ما شاء الله تبارك الله! شايف لك تقدم ملحوظ ومبهر مقارنة بأول، هذا يثبت اجتهادك وحرصك الدائم إنك تطوّر نفسك للأفضل."
   النص الأصلي المشكول: "مَا شَاءَ اللَّهُ! أُلَاحِظُ تَقَدُّمًا مَلْحُوظًا فِي مُسْتَوَاكَ مُقَارَنَةً بِالسَّابِقِ. هَذَا يَدُلُّ عَلَى اجْتِهَادِكَ وَحِرْصِكَ الدَّائِمِ عَلَى التَّطَوُّرِ."

14. دَعْوَةٌ لِلْمَزِيدِ مِنَ الْجُهْدِ (صفحة 2)
   النص المنطوق بالسعودي: "أنت طالب ذكي وعندك طاقات وقدرات عالية ما شاء الله، ومثلك بشوية تركيز زيادة يجيب الدرجة الكاملة بكل ثقة بإذن الله، شد حيلك!"
   النص الأصلي المشكول: "أَنْتَ طَالِبٌ ذَكِيٌّ وَتَمْلِكُ قُدُرَاتٍ كَبِيرَةً، وَأَنَا مُتَأَكِّدٌ أَنَّكَ بِقَلِيلٍ مِنَ التَّرْكِيزِ الْإِضَافِيِّ سَتُحَقِّقُ الْعَلَامَةَ الْكَامِلَةَ، بِإِذْنِ اللَّهِ."

15. تَطَوُّرٌ مُسْتَمِرٌّ (صفحة 2)
   النص المنطوق بالسعودي: "والله إني مبسوط وفخور بالتقدم اللي قاعد تسويه يوم بعد يوم، واصل على هالشغل الطيب فأنت قدوة وقدوة تشرّف لزملائك!"
   النص الأصلي المشكول: "أَنَا سَعِيدٌ جِدًّا بِمُسْتَوَى التَّقَدُّمِ الَّذِي تُحْرِزُهُ يَوْمًا بَعْدَ يَوْمٍ، وَاصِلْ هَذَا الْجُهْدَ الرَّائِعَ فَأَنْتَ مِثَالٌ يُحْتَذَى بِهِ."

16. شُكْرٌ وَتَقْدِيرٌ (صفحة 2)
   النص المنطوق بالسعودي: "ألف شكر على حرصك وانضباطك العالي، مشاركتك تعطي الفصل جو حلو وتحمّس أخوياك، ما شاء الله تبارك الله عليك!"
   النص الأصلي المشكول: "شُكْرًا لِحِرْصِكَ وَانْضِبَاطِكَ. مُشَارَكَتُكَ الدَّائِمَةُ تُضْفِي رَوْنَقًا خَاصًّا عَلَى الْفَصْلِ، وَتُحَفِّزُ زُمَلَاءَكَ، مَا شَاءَ اللَّهُ عَلَيْكَ!"
